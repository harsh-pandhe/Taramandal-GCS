# Taramandal monorepo — single dependency set for all Python packages.
# Install once at the repo root: pip install -r requirements.txt

# Web/API layer (tara-gcs, tara-studio)
fastapi
uvicorn
pydantic
python-multipart
websockets

# Fleet comms — real PX4/hardware backend only (tara-gcs)
# mavsdk 4.0 replaced the gRPC-based Python API (mavsdk.offboard, System, ...)
# with a native binding that drops those modules; drone_manager.py still
# targets the pre-4.0 API, so pin below the break.
mavsdk<4

# Numerics: trajectory math, cKDTree proximity, Hungarian assignment,
# and the vectorized swarm simulator (tara-core, tara-sim)
numpy
scipy
pandas

# Geometry ingest for the Studio voxelizer: font glyph outlines (matplotlib),
# SVG path parsing (svgpathtools) and .obj/.stl meshes (trimesh)
matplotlib
svgpathtools
trimesh

# Ed25519 show-file signatures: signed by `tara sign` (and later the Studio),
# verified by the GCS before any trajectory upload is accepted (tara-core, issue #49)
cryptography

# Orchestrator CLI (apps/tara)
typer

# Process CPU/RSS sampling for tools/benchmark_fleet_scale.py
psutil

# Tests
pytest
pytest-asyncio
# starlette >=1.0 moved TestClient's HTTP backend from httpx to httpx2 and kept
# it an optional extra, so an unpinned `fastapi` install never pulls it in and
# `fastapi.testclient` raises at import. Listed explicitly rather than pinning
# fastapi/starlette back to a pre-1.0 pair, which would hold the runtime on an
# old release to satisfy a test-only dependency.
httpx2
